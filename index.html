专为执行器场景设计，兼顾灵活性与易用性。

多 SDK 集成

原生支持 Junkie、Panda Auth、Luarmor，一键切换验证后端。

-- Patriot Key System · Junkie SDK 完整示例
local PatriotUi = require(script.Parent.PatriotUi)

-- 1. 配置 Junkie SDK
PatriotUi:setSdk("Junkie", {
    apiKey = "your_junkie_api_key",
    timeout = 30,
    endpoint = "https://api.junkie.com/v2/verify",
    retryCount = 3
})

-- 2. 设置 UI 主题（可选）
PatriotUi:setTheme({
    primary = "#1e88e5",
    background = "#0a0c10",
    accent = "#64b5f6"
})

-- 3. 显示 UI 并处理回调
PatriotUi:show({
    onSuccess = function(userData)
        print("✅ 验证成功！用户:", userData.username)
        -- 解锁功能
        unlockFeatures()
    end,
    onFailure = function(errorCode, message)
        print("❌ 验证失败 [", errorCode, "]: ", message)
        -- 处理错误
    end,
    onClose = function()
        print("👋 UI 已关闭")
    end
})
-- Patriot Key System · Panda Auth 完整示例
local PatriotUi = require(script.Parent.PatriotUi)

-- 1. 配置 Panda Auth
PatriotUi:setSdk("Panda", {
    apiKey = "panda_public_key",
    secretKey = "panda_secret_key",
    timeout = 20,
    sandbox = true  -- 沙箱模式
})

-- 2. 设置无密钥模式（开发用）
-- PatriotUi:setKeylessMode(true)

-- 3. 显示 UI
PatriotUi:show({
    onSuccess = function()
        print("✅ Panda Auth 验证通过")
        -- 执行器解锁
    end,
    onFailure = function(err)
        print("❌ 失败:", err)
    end
})
-- Patriot Key System · Luarmor API 完整示例
local PatriotUi = require(script.Parent.PatriotUi)

-- 1. 配置 Luarmor
PatriotUi:setSdk("Luarmor", {
    apiKey = "luarmor_api_key",
    productId = "your_product_id",
    version = "v1",
    encryption = true
})

-- 2. 自定义验证逻辑（可选）
PatriotUi:setCustomValidation(function(key)
    -- 额外的本地验证
    return #key > 8
end)

-- 3. 显示 UI
PatriotUi:show({
    onSuccess = function(license)
        print("✅ Luarmor 验证成功，许可证:", license)
    end,
    onFailure = function(code, msg)
        print("❌ 失败:", code, msg)
    end
})
